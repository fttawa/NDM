cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
cmake_policy(SET CMP0144 NEW)
project(ndm_bt LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_MINSIZEREL ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(static_runtime ON CACHE BOOL "" FORCE)
set(build_tests OFF CACHE BOOL "" FORCE)
set(build_examples OFF CACHE BOOL "" FORCE)
set(build_tools OFF CACHE BOOL "" FORCE)
set(python-bindings OFF CACHE BOOL "" FORCE)
set(deprecated-functions OFF CACHE BOOL "" FORCE)
set(logging OFF CACHE BOOL "" FORCE)
set(i2p OFF CACHE BOOL "" FORCE)
set(webtorrent OFF CACHE BOOL "" FORCE)
set(mutable-torrents OFF CACHE BOOL "" FORCE)
set(streaming OFF CACHE BOOL "" FORCE)
set(OPENSSL_USE_STATIC_LIBS ON)
set(Boost_NO_BOOST_CMAKE ON)
add_subdirectory("${NDM_LIBTORRENT_SOURCE}" libtorrent EXCLUDE_FROM_ALL)
set_source_files_properties(engine.cpp fixture.cpp PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/common.hpp")
foreach(target ndm-bt-engine ndm-bt-fixture)
  if(target STREQUAL "ndm-bt-engine")
    add_executable(${target} engine.cpp)
  else()
    add_executable(${target} fixture.cpp)
  endif()
  target_link_libraries(${target} PRIVATE torrent-rasterbar)
  target_include_directories(${target} PRIVATE "${NDM_JSON_INCLUDE}")
  target_compile_options(${target} PRIVATE /utf-8 /W4 /Gy /Gw)
  target_link_options(${target} PRIVATE /OPT:REF /OPT:ICF)
endforeach()
